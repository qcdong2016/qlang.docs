# 19. 函数


### 19.1 声明

```
package demo

func empty() {}

func add(a int, b int) int {
    return a + b
}

func add2(a, b int) int {
    return a + b
}

func swap(a int, b int) (int, int) {
    return b, a
}

func Open(path string) (int, error) {
    return 0, nil
}

func Max(a int, b int) int {
    if a > b { return a }
    return b
}
```

### 19.2 调用

```
package demo

func add(a int, b int) int { return a + b }
func swap(a int, b int) (int, int) { return b, a }

func f() {
    x := add(1, 2)
    a, b := swap(1, 2)
    _, c := swap(3, 4)
    _ = x
    _ = a
    _ = b
    _ = c
}
```

### 19.3 可变参数

```
package demo

func sum(xs ...int) int {
    n := 0
    for i := 0; i < len(xs); i += 1 {
        n += xs[i]
    }
    return n
}

func f() {
    a := sum(1, 2, 3)
    var b []int = []int{4, 5}
    c := sum(b...)
}
```

### 19.4 函数值与闭包

```
package demo

struct Node : Ref { v int };

func apply(f func(int) int, x int) int {
    return f(x)
}

func f() {
    add := func(a int, b int) int { return a + b }
    inc := func(x int) int { return x + 1 }
    n := apply(inc, 1)
    k := 10
    g := func(x int) int { return x + k }
    node := Node{v: 1}
    h := func() int { return node.v }
}
```

捕获值：拷贝。捕获 Ref 结构体：`retain`。

```
package demo

func f() {
    k := 10
    g := func(x int) int { return x + k }
    _ = g(1)
}
```
