# 24. `error`


```
package demo

interface error {
    Error() string
};

struct Err : Ref {
    msg string
};

func Err.Error() string { return this.msg }

func Open(path string) (int, error) {
    if path == "" {
        return (0, Err{msg: "empty"})
    }
    return (1, nil)
}

func f() {
    r := Open("a")
    fd := r.0
    err := r.1
    if err != nil {
        panic(err.Error())
    }
    _ = fd
}
```
