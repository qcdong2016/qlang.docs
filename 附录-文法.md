# 附录-文法

```
SourceFile    = "package" identifier { ImportDecl | Declaration } .

ImportDecl    = "import" ( ImportSpec | "(" { ImportSpec } ")" ) .
ImportSpec    = [ identifier | "." ] string_lit .

Declaration   = LetDecl | TypeAlias | StructDecl | InterfaceDecl | FuncDecl .

LetDecl       = "let" ( LetSpec | "(" { LetSpec } ")" ) .
LetSpec       = IdentifierList [ ":" Type ] [ "=" ExpressionList ] .
TypeAlias     = "type" identifier "=" Type .

StructDecl    = "struct" identifier [ TypeParams ] [ ":" "Ref" ] "{" { FieldDecl } "}" ";" .
TypeParams    = "<" identifier ":" identifier { "," identifier ":" identifier } ">" .
FieldDecl     = IdentifierList ":" Type .

InterfaceDecl = "interface" identifier "{" { MethodSpec | identifier } "}" ";" .
MethodSpec    = identifier Signature .

FuncDecl      = "func" [ identifier "." ] ( identifier | operator ) [ TypeParams ] Signature Block .
Signature     = Parameters [ Result ] .
Parameters    = "(" [ ParamList ] ")" .
ParamList     = Param { "," Param } .
Param         = IdentifierList ":" [ "..." ] Type .
Result        = ":" Type | ":" "(" Type { "," Type } ")" .
operator      = "+" | "-" | "*" | "/" | "%" | "[]" | "[]="
              | "&" | "|" | "^" | "&^" | "<<" | ">>"
              | "==" | "!=" | "<" | "<=" | ">" | ">=" | "!" .

Type          = identifier [ TypeArgs ]
              | "[" int_lit "]" Type
              | "[" "]" Type
              | "Map" TypeArgs
              | "func" Signature
              | "weak" Type
              | "any" .
TypeArgs      = "<" Type { "," Type } ">" .

Block         = "{" { Statement } "}" .

Statement     = Declaration | SimpleStmt | Block | IfStmt | ForStmt
              | SwitchStmt | "break" [ identifier ]
              | "continue" [ identifier ] | "goto" identifier
              | ReturnStmt | DeferStmt | identifier ":" Statement .
SimpleStmt    = Expression | Assignment | ShortVarDecl .
Assignment    = ExpressionList assign_op ExpressionList .
ShortVarDecl  = IdentifierList ":=" ExpressionList .
ReturnStmt    = "return" [ ExpressionList ] .
DeferStmt     = "defer" Expression .

IfStmt        = "if" [ SimpleStmt ";" ] Expression Block [ "else" ( IfStmt | Block ) ] .
ForStmt       = "for" ( [ ForClause ] | [ Expression ] | RangeClause ) Block .
ForClause     = [ SimpleStmt ] ";" [ Expression ] ";" [ SimpleStmt ] .
RangeClause   = [ ExpressionList "=" | IdentifierList ":=" ] "range" Expression .
SwitchStmt    = "switch" [ SimpleStmt ";" ] [ Expression | "typeof" "(" Expression ")" ]
                "{" { CaseClause } "}" .
CaseClause    = ( "case" ExpressionList | "case" Type { "," Type } | "default" ) ":" { Statement } .

Expression    = UnaryExpr { binary_op UnaryExpr } .
UnaryExpr     = [ unary_op ] PrimaryExpr .
PrimaryExpr   = Operand { "." identifier [ TypeArgs ] | Index | Slice | TypeArgs | Arguments | CompositeTail } .
Operand       = identifier | literal | Type | "(" Expression ")" | FunctionLit .
FunctionLit   = "func" Signature Block .
CompositeTail = "{" [ LiteralValue ] "}" .
Arguments     = "(" [ ExpressionList [ "..." ] ] ")" .
Index         = "[" Expression "]" .
Slice         = "[" [ Expression ] ":" [ Expression ] "]" .
```

`CompositeTail` 只接在类型之后：`Point{...}`、`A<T>{...}`、`[3]int{...}`、`[]int{...}`、`Map<K,V>{...}`。`if x {` 的 `{` 是 `Block`。

调用与字面量可显式写实参：`B<T>()`、`a.func1<T>()`、`A<T>{}`。

类型位置的 `<...>` 归 `TypeArgs`；表达式位置的 `<` `>` 归 `binary_op`。
